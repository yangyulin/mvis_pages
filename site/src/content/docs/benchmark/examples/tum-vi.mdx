## Config and run

```yaml title="tum_vi_imu1_cam_imu.yaml" {9,20-24,28,33,37-39}
input:
  format: ros1
  path: dataset-calib-imu1.bag
  cam_topics:
  - /cam0/image_raw
  - /cam1/image_raw
  imu_topics:
  - /imu0
  window_mode: motion
  segment_length: 80.0
board:
  board_type: apriltag
  rows: 6
  cols: 6
  tag_size: 0.088
  tag_spacing: 0.3
  tag_family: t36h11
  heights_file: null
camera_input:
  path: camera_prior_kb4.yaml
  cameras:
  - estimate_readout: false
  - estimate_readout: false
    factor_hz: 8.0
solve:
  max_iterations: 40
  corner_subsample: all
  estimate_t_ga: true
  sigma_t_ga: 0.01
  vision_pose: imu
estimate:
  camera:
    intrinsics: true
    tr: false
    td: true
    extrinsics: true
  imu:
    intrinsics: dw_da
    model: calyx
output:
  path: cam_imu.json
```

```bash title="shell"
mvis cam-imu -c tum_vi_imu1_cam_imu.yaml
```
