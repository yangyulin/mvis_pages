## Config and run

```yaml title="looper_insight9_bag1_cam_imu.yaml" {22,24-28,35,40}
input:
  format: euroc
  path: 1/mav0
  cam_topics:
  - /cam0
  - /cam1
  - /cam2
  imu_topics:
  - /imu0
  window_mode: manual
  t0: 0.0
  window: 0.0
board:
  board_type: apriltag
  rows: 6
  cols: 6
  tag_size: 0.055
  tag_spacing: 0.3
  tag_family: t36h11
camera_input:
  path: insight9_seed_calib.yaml
  free_intrinsics: true
  estimate_readout: false
  cameras:
  - {}
  - {}
  - estimate_readout: true
    rolling_mode: 0
solve:
  state_hz: 30.0
  max_iterations: 40
  imu_intrinsics: dw_da
  imu_intrinsics_model: calyx
  corner_subsample: top_left
  estimate_t_ga: true
  sigma_t_ga: 0.01
  init_kwargs:
    keyframe_hz: 2.0
    scale_rel_tol: 0.3
  sigma_g: 0.00021
  sigma_a: 0.0032
  sigma_bg: 0.001
  sigma_ba: 0.002
output:
  path: cam_imu.json
```
