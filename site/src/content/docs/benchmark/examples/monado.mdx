## Config and run

```yaml title="monado_mgc_imucalib01_cam_imu.yaml" {9,15,21-24,29,34}
input:
  format: euroc
  path: MGC04_imucalib01_1/mav0
  cam_topics:
  - cam0
  - cam1
  imu_topics:
  - imu0
  window_mode: motion
  segment_length: 80.0
board:
  board_type: apriltag
  rows: 6
  cols: 6
  tag_size: 0.03
  tag_spacing: 0.3
  tag_family: t36h11
  heights_file: null
camera_input:
  path: camera_prior.yaml
  cameras:
  - estimate_readout: false
  - estimate_readout: false
    factor_hz: 8.0
solve:
  max_iterations: 40
  pnp_method: ippe
  corner_subsample: all
  estimate_t_ga: true
  sigma_t_ga: 0.01
  vision_pose: imu
estimate:
  camera:
    intrinsics: true
    tr: false
    td: true
    extrinsics: true
  imu:
    intrinsics: dw_da
    model: calyx
output:
  path: cam_imu.json
```

```bash title="shell"
mvis cam-imu -c monado_mgc_imucalib01_cam_imu.yaml
```
