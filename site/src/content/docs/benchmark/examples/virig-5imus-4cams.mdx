## Config and run

```yaml title="virig_5imus_4cams_cam_imu.yaml" {9-15,25-30,32,37,46}
input:
  format: ros1
  path: 5imus_4cams_01.bag
  cam_topics:
  - /t265_01/fisheye1/image_raw
  - /t265_01/fisheye2/image_raw
  - /t265_03/fisheye1/image_raw
  - /t265_03/fisheye2/image_raw
  imu_topics:
  - /mti/sensor/imu_raw
  - /gx3_25/data
  - /gx3_35/imu/data
  - /t265_01/imu
  - /t265_03/imu
  window_mode: motion
  segment_length: 80.0
board:
  board_type: apriltag
  rows: 6
  cols: 6
  tag_size: 0.088
  tag_spacing: 0.3
  tag_family: t36h11
camera_input:
  path: camera_prior.yaml
  cameras:
  - {}
  - factor_hz: 4.0
  - factor_hz: 4.0
  - factor_hz: 4.0
solve:
  estimate_t_ga: true
  max_iterations: 40
  corner_subsample: two_anti
estimate:
  camera:
    intrinsics: true
    td: true
    extrinsics: true
    tr: false
  imu:
    intrinsics: dw_da
    model: calyx
    extrinsics: true
    td: true
  board_points: true
output:
  path: cam_imu.json
```

```bash title="shell"
mvis cam-imu -c virig_5imus_4cams_cam_imu.yaml
```
