## Config and run

```yaml title="virig_4imus_4cams_cam_imu.yaml" {9-14,26-32,39,44,46}
input:
  format: ros1
  path: 4imus_4cams_01.bag
  cam_topics:
  - /t265/fisheye1/image_raw
  - /t265/fisheye2/image_raw
  - /elp/split_sync_image_node/left/image_raw
  - /elp/split_sync_image_node/right/image_raw
  imu_topics:
  - /imu/data
  - /gx3_25/data
  - /gx3_35/imu/data
  - /t265/imu
  window_mode: motion
  segment_length: 80.0
board:
  board_type: apriltag
  rows: 6
  cols: 6
  tag_size: 0.088
  tag_spacing: 0.3
  tag_family: t36h11
  heights_file: board_shape_4imus_4cams.yaml
camera_input:
  path: camera_prior.yaml
  cameras:
  - {}
  - factor_hz: 4.0
  - estimate_readout: true
    factor_hz: 4.0
  - estimate_readout: true
    factor_hz: 4.0
solve:
  max_iterations: 40
  corner_subsample: two_anti
  estimate_t_ga: true
  sigma_t_ga: 0.01
  vision_pose: imu
  sigma_g: 0.00065
  sigma_a: 0.0026
  sigma_bg: 0.001
  sigma_ba: 0.002
estimate:
  board_points: true
  camera:
    intrinsics: true
    tr: false
    td: true
    extrinsics: true
  imu:
    intrinsics: dw_da
    model: calyx
    extrinsics: true
    td: true
output:
  path: cam_imu.json
```

```bash title="shell"
mvis cam-imu -c virig_4imus_4cams_cam_imu.yaml
```
